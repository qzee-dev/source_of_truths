For a truly reproducible EC2 environment, import all items in the checklist above. The first six resources are the minimum required for your current EC2 instance.

1. EC2 Instance :
 aws ec2 describe-instances \
  --instance-ids i-076cb0a507884f3a6

2. EBS Volumes: 
aws ec2 describe-volumes \
  --filters Name=attachment.instance-id,Values=i-076cb0a507884f3a6

3. Network Interfaces:
aws ec2 describe-network-interfaces \
  --filters Name=attachment.instance-id,Values=i-076cb0a507884f3a6

4. Security Groups
aws ec2 describe-security-groups \
  --group-ids sg-001adec0cd944bbb2

5. Subnet: 
aws ec2 describe-subnets \
  --subnet-ids subnet-0ccf38b701d16ffc

6. VPC
aws ec2 describe-vpcs \
  --vpc-ids vpc-0b69e283891dc7505


7. Route Tables:
aws ec2 describe-route-tables \
  --filters Name=vpc-id,Values=vpc-0b69e283891dc7505

8. Internet Gateway:
aws ec2 describe-internet-gateways \
  --filters Name=attachment.vpc-id,Values=vpc-0b69e283891dc7505

9. Elastic IP:
aws ec2 describe-addresses

10. Key Pair
aws ec2 describe-key-pairs \
  --key-names server-01

11. IAM Instance Profile
aws ec2 describe-instances \
  --instance-ids i-076cb0a507884f3a6 \
  --query "Reservations[].Instances[].IamInstanceProfile"

12. NACLs
aws ec2 describe-network-acls \
  --filters Name=vpc-id,Values=vpc-0b69e283891dc7505


13. DHCP Options
aws ec2 describe-vpcs \
  --vpc-ids vpc-0b69e283891dc7505 \
  --query "Vpcs[].DhcpOptionsId"



aws ec2 describe-dhcp-options


14. Tags
aws ec2 describe-tags \
  --filters Name=resource-id,Values=i-076cb0a507884f3a6

====================
Solution
==================
Terraform/
├── provider.tf
├── main.tf
├── imports.tf
├── generated.tf





[provider.tf:]
terraform {
  required_version = ">= 1.5"

  required_providers {
    aws = {
      source  = "hashicorp/aws"
      version = "~> 6.0"
    }
  }
}

provider "aws" {
  region = "eu-north-1"
}


[main.tf:]
resource "aws_instance" "server01" {}

resource "aws_vpc" "main" {}

resource "aws_subnet" "public" {}

resource "aws_security_group" "server01" {}

resource "aws_network_interface" "server01" {}

resource "aws_ebs_volume" "root" {}


[imports.tf]

import {
  to = aws_instance.server01
  id = "i-076cb0a507884f3a6"
}

import {
  to = aws_vpc.main
  id = "vpc-0b69e283891dc7505"
}

import {
  to = aws_subnet.public
  id = "subnet-0ccf38b701d16ffc"
}

import {
  to = aws_security_group.server01
  id = "sg-001adec0cd944bbb2"
}

import {
  to = aws_network_interface.server01
  id = "eni-05804cd84d5d70126"
}

import {
  to = aws_ebs_volume.root
  id = "vol-0ae35254f31050269"
}

Commands:
terraform init
terraform plan -generate-config-out=generated.tf

------------------------------------
Once generated.tf looks correct:
--------------------------------------

Copy the generated resources into main.tf.
Delete the empty resource blocks.
Delete imports.tf (optional).
Run:
Shell
terraform plan

Expected result:

Plain Text
No changes. Infrastructure matches the configuration.
